cmake_minimum_required(VERSION 3.16)
project(seedless_config_tests C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_executable(test_config ../../src/config.c ../../src/config_json.c test_config.c)
target_include_directories(test_config PRIVATE ../../src)
add_executable(config_probe ../../src/config.c ../../src/config_json.c ../../src/launch_options.c config_probe.c)
target_include_directories(config_probe PRIVATE ../../src)
add_executable(test_launch_options ../../src/launch_options.c test_launch_options.c)
target_include_directories(test_launch_options PRIVATE ../../src)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
enable_testing()
add_test(NAME seedless-config COMMAND test_config)
add_test(NAME seedless-launch-options COMMAND test_launch_options)
add_test(NAME seedless-config-cli COMMAND ${CMAKE_COMMAND} -E env CONFIG_PROBE=$<TARGET_FILE:config_probe> ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/../test_config_cli.py)
